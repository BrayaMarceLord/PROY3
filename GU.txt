library IEEE;

use IEEE.STD_LOGIC_1164.ALL;

use IEEE.NUMERIC_STD.ALL;



entity GU is

    Port (

        clk      : in  STD_LOGIC;

        data     : in  STD_LOGIC_VECTOR(3 downto 0);

        hcount   : in  STD_LOGIC_VECTOR(10 downto 0);

        vcount   : in  STD_LOGIC_VECTOR(9 downto 0);

        rgb      : out STD_LOGIC_VECTOR(11 downto 0);

        dir_b    : out STD_LOGIC_VECTOR(12 downto 0) -- Actualizado a 13 bits

    );

end GU;



architecture Behavioral of GU is



    signal hc      : unsigned(10 downto 0);

    signal vc      : unsigned(9 downto 0);



    signal h_off1  : unsigned(10 downto 0);

    signal h_off2  : unsigned(10 downto 0);

    signal v_off   : unsigned(9 downto 0);



    signal col1    : unsigned(6 downto 0); -- 7 bits (0 a 127 columnas)

    signal col2    : unsigned(6 downto 0); -- 7 bits (0 a 127 columnas)

    signal row     : unsigned(5 downto 0); -- 6 bits (0 a 63 filas)



    signal in_t1   : std_logic;

    signal in_t2   : std_logic;



    -- Registros para compensar latencia de 1 ciclo de la memoria BRAM

    signal in_t1_r : std_logic := '0';

    signal in_t2_r : std_logic := '0';



begin



    hc <= unsigned(hcount);

    vc <= unsigned(vcount);



    -- =====================================================

    -- ZONAS DE PANTALLA MULTICORE (128x64 celdas)

    -- Tamaño de cada "grano" de arena: 2x2 pixeles

    -- =====================================================



    -- Tablero 1: Cantidades de Arena (Izquierda)

    -- x: 32 a 287 (Ancho 256), y: 176 a 303 (Alto 128)

    in_t1 <= '1' when (hc >= 32 and hc < 288) and (vc >= 176 and vc < 304) else '0';



    -- Tablero 2: Destellos de Avalancha (Derecha)

    -- x: 352 a 607 (Ancho 256), y: 176 a 303 (Alto 128)

    in_t2 <= '1' when (hc >= 352 and hc < 608) and (vc >= 176 and vc < 304) else '0';



    process(clk)

    begin

        if rising_edge(clk) then

            in_t1_r <= in_t1;

            in_t2_r <= in_t2;

        end if;

    end process;



    -- =====================================================

    -- CÁLCULO DE OFFSETS

    -- =====================================================

    h_off1 <= hc - 32  when in_t1 = '1' else (others => '0');

    h_off2 <= hc - 352 when in_t2 = '1' else (others => '0');

    v_off  <= vc - 176 when (in_t1 = '1' or in_t2 = '1') else (others => '0');



    -- Escalamiento 2x2: Dividimos el offset físico desplazando 1 bit a la derecha.

    col1 <= h_off1(7 downto 1);

    col2 <= h_off2(7 downto 1);

    row  <= v_off(6 downto 1); 



    -- =====================================================

    -- DIRECCIÓN UNIFICADA A LA MEMORIA (13 BITS)

    -- =====================================================

    dir_b <= std_logic_vector(row & col1) when in_t1 = '1'

             else std_logic_vector(row & col2) when in_t2 = '1'

             else (others => '0');



    -- =====================================================

    -- DIBUJADO DE COLORES

    -- =====================================================

    process(in_t1_r, in_t2_r, data)

    begin

        if in_t1_r = '1' then

            -- data(3 downto 1) contiene la cantidad de granos

            case data(3 downto 1) is

                when "000"  => rgb <= "000000000000"; -- Negro (0 granos)

                when "001"  => rgb <= "000000001111"; -- Azul (1 grano)

                when "010"  => rgb <= "000011110000"; -- Verde (2 granos)

                when "011"  => rgb <= "111100000000"; -- Rojo (3 granos)

                when "100"  => rgb <= "111111110000"; -- Amarillo (Toppling transitorio)

                when "101"  => rgb <= "111111110000"; 

                when "110"  => rgb <= "111111110000"; 

                when "111"  => rgb <= "111111110000"; 

                when others => rgb <= "000000000000";

            end case;



        elsif in_t2_r = '1' then

            -- data(0) es el flag de avalancha

            if data(0) = '1' then

                rgb <= "111111111111"; -- Blanco puro (Avalancha activa)

            else

                rgb <= "000000000000"; -- Negro (Silencio)

            end if;



        else

            rgb <= "000000000000"; -- Fondo negro fuera de los tableros

        end if;

    end process;



end Behavioral;
