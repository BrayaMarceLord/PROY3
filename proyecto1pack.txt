

library IEEE;

use IEEE.STD_LOGIC_1164.ALL;



package proyecto1pack is



component ASM is

    Port (

        clk : in STD_LOGIC; ini : in STD_LOGIC; reset : in STD_LOGIC;

        we : out STD_LOGIC_VECTOR(0 downto 0); ab : out STD_LOGIC_VECTOR(12 downto 0);

        dbi : out STD_LOGIC_VECTOR(3 downto 0); dbo : in STD_LOGIC_VECTOR(3 downto 0);

        px_offset : in STD_LOGIC; EnaN : in STD_LOGIC; CtrlV : in STD_LOGIC; 

        DesC : in STD_LOGIC;  -- 0=Centro, 1=Aleatorio

        cinit : in STD_LOGIC; -- 0=Limpio, 1=Ruido inicial

        rx : in STD_LOGIC_VECTOR(5 downto 0); ry : in STD_LOGIC_VECTOR(5 downto 0); rval : in STD_LOGIC_VECTOR(1 downto 0)

    );

end component;



component MAE_Multicore is

    Port (

        clk : in STD_LOGIC; ini : in STD_LOGIC; reset : in STD_LOGIC;

        sw_core1 : in STD_LOGIC_VECTOR(2 downto 0); sw_core2 : in STD_LOGIC_VECTOR(2 downto 0);

        cinit : in STD_LOGIC; 

        dir_b : in STD_LOGIC_VECTOR(12 downto 0); data : out STD_LOGIC_VECTOR(3 downto 0)

    );

end component;



component PRNG is

    Generic (SEED : std_logic_vector(7 downto 0) := "10101010");

    Port ( clk : in STD_LOGIC; random_out : out std_logic_vector(1 downto 0); rand_x : out std_logic_vector(5 downto 0); rand_y : out std_logic_vector(5 downto 0) );

end component;



component mem is

    PORT ( clka : IN STD_LOGIC; wea : IN STD_LOGIC_VECTOR(0 DOWNTO 0); addra : IN STD_LOGIC_VECTOR(12 DOWNTO 0); dina : IN STD_LOGIC_VECTOR(3 DOWNTO 0); douta : OUT STD_LOGIC_VECTOR(3 DOWNTO 0); clkb : IN STD_LOGIC; web : IN STD_LOGIC_VECTOR(0 DOWNTO 0); addrb : IN STD_LOGIC_VECTOR(12 DOWNTO 0); dinb : IN STD_LOGIC_VECTOR(3 DOWNTO 0); doutb : OUT STD_LOGIC_VECTOR(3 DOWNTO 0) );

end component;



component GU is

    Port ( clk : in STD_LOGIC; data : in STD_LOGIC_VECTOR(3 downto 0); hcount : in STD_LOGIC_VECTOR(10 downto 0); vcount : in STD_LOGIC_VECTOR(9 downto 0); rgb : out STD_LOGIC_VECTOR(11 downto 0); dir_b : out STD_LOGIC_VECTOR(12 downto 0) );

end component;



component vgac is

    Port ( clk : in STD_LOGIC; clk_25_out : out STD_LOGIC; reset : in STD_LOGIC; hsync : out STD_LOGIC; vsync : out STD_LOGIC; dir : out STD_LOGIC_VECTOR (12 downto 0); hcount : out STD_LOGIC_VECTOR (10 downto 0); vcount : out STD_LOGIC_VECTOR (9 downto 0) );

end component;



end proyecto1pack;
