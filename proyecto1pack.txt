library IEEE;

use IEEE.STD_LOGIC_1164.ALL;



package proyecto1pack is



-- ASM

component ASM is

    Port (

        clk   : in  STD_LOGIC;

        ini   : in  STD_LOGIC;

        reset : in  STD_LOGIC;

        we    : out STD_LOGIC_VECTOR(0 downto 0);

        ab    : out STD_LOGIC_VECTOR(11 downto 0);

        dbi   : out STD_LOGIC_VECTOR(3 downto 0);

        dbo   : in  STD_LOGIC_VECTOR(3 downto 0);

        cinit : in  STD_LOGIC;

        op    : in  STD_LOGIC;

        rx    : in  STD_LOGIC_VECTOR(5 downto 0);

        ry    : in  STD_LOGIC_VECTOR(5 downto 0);

        rval  : in  STD_LOGIC_VECTOR(1 downto 0)

    );

end component;



-- MAE

component MAE is

    Port (

        clk    : in  STD_LOGIC;

        clk_25 : in  STD_LOGIC;

        ini    : in  STD_LOGIC;

        reset  : in  STD_LOGIC;

        dir_b  : in  STD_LOGIC_VECTOR(11 downto 0);

        data   : out STD_LOGIC_VECTOR(3 downto 0);

        bt     : in  STD_LOGIC_VECTOR(1 downto 0)

    );

end component;



-- GU

component GU is

    Port (

        clk    : in  STD_LOGIC;

        data   : in  STD_LOGIC_VECTOR(3 downto 0);

        hcount : in  STD_LOGIC_VECTOR(10 downto 0);

        vcount : in  STD_LOGIC_VECTOR(9 downto 0);

        rgb    : out STD_LOGIC_VECTOR(11 downto 0);

        dir_b  : out STD_LOGIC_VECTOR(11 downto 0)

    );

end component;



-- MEMI (Dual Port RAM unificada)

component memi is

    Port (

        clka  : in  STD_LOGIC;

        wea   : in  STD_LOGIC_VECTOR(0 downto 0);

        addra : in  STD_LOGIC_VECTOR(11 downto 0);

        dina  : in  STD_LOGIC_VECTOR(3 downto 0);

        douta : out STD_LOGIC_VECTOR(3 downto 0);

        clkb  : in  STD_LOGIC;

        addrb : in  STD_LOGIC_VECTOR(11 downto 0);

        doutb : out STD_LOGIC_VECTOR(3 downto 0)

    );

end component;



-- VGAC

component vgac is

    Port (

        clk    : in  STD_LOGIC;
		  clk_25_out: out  STD_LOGIC;

        reset  : in  STD_LOGIC;

        hsync  : out STD_LOGIC;

        vsync  : out STD_LOGIC;

        dir    : out STD_LOGIC_VECTOR (11 downto 0);

        hcount : out STD_LOGIC_VECTOR (10 downto 0);

        vcount : out STD_LOGIC_VECTOR (9 downto 0)

    );

end component;



end proyecto1pack;
