library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity MAE is
    Port (
        clk    : in  STD_LOGIC;
        clk_25 : in  STD_LOGIC;
        ini    : in  STD_LOGIC;
        reset  : in  STD_LOGIC;
        dir_b  : in  STD_LOGIC_VECTOR(11 downto 0);
        data   : out STD_LOGIC_VECTOR(3 downto 0);
        bt     : in  STD_LOGIC_VECTOR(1 downto 0)
    );
end MAE;

architecture Behavioral of MAE is

signal we  : std_logic_vector(0 downto 0);
signal ab  : std_logic_vector(11 downto 0);
signal dbi : std_logic_vector(3 downto 0);
signal dbo : std_logic_vector(3 downto 0);

signal rx      : std_logic_vector(5 downto 0);
signal ry      : std_logic_vector(5 downto 0);
signal rval    : std_logic_vector(1 downto 0);


begin

-- ASM (El Cerebro de Control)
c0: entity work.ASM
port map (
    clk   => clk,
    ini   => ini,
    reset => reset,

    we  => we,
    ab  => ab,
    dbi => dbi,
    dbo => dbo,

    -- PUENTE DE LOS SWITCHES FÍSICOS
    cinit => bt(0),
    op    => bt(1),

    rx    => rx,
    ry    => ry,
    rval  => rval
   
);

-- MEMORIA UNIFICADA (Generada por IP Core - Opción 2)
-- bits 3:1 = valor de granos, bit 0 = flag de avalancha
c1: entity work.memi
port map (
    -- Puerto A: Interfaz con el ASM (Lectura/Escritura)
    clka  => clk,
    wea   => we,
    addra => ab,
    dina  => dbi,
    douta => dbo,
    
    -- Puerto B: Interfaz con la Unidad de Gráficos (Solo Lectura)
    clkb  => clk_25,
    web   => "0",        -- Desactivamos la escritura en el puerto B
    addrb => dir_b,
    dinb  => "0000",     -- Datos de entrada nulos (no se usan)
    doutb => data
);

-- PRNG (Generador Pseudoaleatorio)
c3: entity work.PRNG
port map (
    clk   => clk,
    reset => reset,
   
    rx    => rx,
    ry    => ry,
    rval  => rval
);

end Behavioral;
