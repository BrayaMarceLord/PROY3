library IEEE;

use IEEE.STD_LOGIC_1164.ALL;

use IEEE.NUMERIC_STD.ALL;



entity proy1 is

    Port (

        clk   : in  STD_LOGIC;

        ini   : in  STD_LOGIC;

        reset : in  STD_LOGIC;

        bt    : in  STD_LOGIC_VECTOR(1 downto 0);

        hsync : out STD_LOGIC;

        vsync : out STD_LOGIC;

        rgb   : out STD_LOGIC_VECTOR(11 downto 0)

    );

end proy1;



architecture Behavioral of proy1 is



signal hcounti : std_logic_vector(10 downto 0);

signal vcounti : std_logic_vector(9 downto 0);

signal dir_b_i : std_logic_vector(11 downto 0);

signal data_i  : std_logic_vector(3 downto 0);

signal clk_25  : std_logic; -- Señal que viaja desde el VGA al resto



begin



-- MAE (Máquina y Memoria)

c0: entity work.MAE

port map (

    clk    => clk,

    clk_25 => clk_25,  -- Recibe el reloj desde el controlador VGA

    ini    => ini,

    reset  => reset,

    dir_b  => dir_b_i,

    data   => data_i,

    bt     => bt

);



-- CONTROLADOR VGA (Ahora contiene el divisor de reloj)

c1: entity work.vgac

port map (

    clk        => clk,         -- Entran 100MHz

    reset      => reset,

    clk_25_out => clk_25,      -- Salen 25MHz

    hsync      => hsync,

    vsync      => vsync,

    dir        => open,

    hcount     => hcounti,

    vcount     => vcounti

);



-- UNIDAD DE GRÁFICOS

c2: entity work.GU

port map (

    clk    => clk_25,          -- Recibe el reloj de 25MHz

    data   => data_i,

    hcount => hcounti,

    vcount => vcounti,

    rgb    => rgb,

    dir_b  => dir_b_i

);



end Behavioral;
