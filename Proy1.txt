library IEEE;

use IEEE.STD_LOGIC_1164.ALL;



entity proy1 is

    Port (

        clk      : in  STD_LOGIC;

        ini      : in  STD_LOGIC;

        reset    : in  STD_LOGIC;

        cinit    : in  STD_LOGIC;

        sw_core1 : in  STD_LOGIC_VECTOR(2 downto 0);

        sw_core2 : in  STD_LOGIC_VECTOR(2 downto 0);

        hsync    : out STD_LOGIC;

        vsync    : out STD_LOGIC;

        rgb      : out STD_LOGIC_VECTOR(11 downto 0)

    );

end proy1;



architecture Behavioral of proy1 is

    -- Señales VGA

    signal hcount_i : std_logic_vector(10 downto 0);

    signal vcount_i : std_logic_vector(9 downto 0);

    

    -- Dirección que el VGA le pasa al GU para calcular

    -- La memoria recibe la dirección desde el GU, no desde el VGAC

    signal dir_mem_i: std_logic_vector(12 downto 0); 

    

    signal data_i   : std_logic_vector(3 downto 0);

begin



    -- 1. VGAC solo genera coordenadas

    Inst_VGAC: entity work.vgac port map (

        clk => clk, reset => reset, hsync => hsync, vsync => vsync, 

        hcount => hcount_i, vcount => vcount_i, dir => open -- Ya no controlamos dir aquí

    );



    -- 2. GU recibe coordenadas y genera la dirección para la RAM

    Inst_GU: entity work.GU port map (

        clk => clk, data => data_i, hcount => hcount_i, 

        vcount => vcount_i, rgb => rgb, dir_b => dir_mem_i

    );



    -- 3. MAE recibe la dirección del GU

    Inst_MAE: entity work.MAE_Multicore port map (

        clk => clk, ini => ini, reset => reset, cinit => cinit,

        sw_core1 => sw_core1, sw_core2 => sw_core2, 

        dir_b => dir_mem_i, data => data_i

    );



end Behavioral;
