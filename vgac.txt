library IEEE;

use IEEE.STD_LOGIC_1164.ALL;

use IEEE.STD_LOGIC_UNSIGNED.ALL;



entity vgac is

    Port (

        clk        : in  STD_LOGIC;

        clk_25_out : out STD_LOGIC;

        reset      : in  STD_LOGIC;

        hsync      : out STD_LOGIC;

        vsync      : out STD_LOGIC;

        dir        : out STD_LOGIC_VECTOR (12 downto 0); -- Actualizado a 13 bits

        hcount     : out STD_LOGIC_VECTOR (10 downto 0);

        vcount     : out STD_LOGIC_VECTOR (9 downto 0)

    );

end vgac;



architecture Behavioral of vgac is

    -- Señales para generar el reloj de 25MHz a partir de los 100MHz (Para 640x480)

    signal clk_25 : std_logic := '0';

    signal clk_div : std_logic_vector(1 downto 0) := "00";

    

    -- Contadores de barrido de pantalla

    signal h_cnt : std_logic_vector(10 downto 0) := (others => '0');

    signal v_cnt : std_logic_vector(9 downto 0) := (others => '0');

begin



    -- Divisor de frecuencia (100MHz a 25MHz)

    process(clk, reset)

    begin

        if reset = '1' then

            clk_div <= "00";

            clk_25 <= '0';

        elsif rising_edge(clk) then

            if clk_div = "01" then

                clk_25 <= not clk_25;

                clk_div <= "00";

            else

                clk_div <= clk_div + 1;

            end if;

        end if;

    end process;

    

    clk_25_out <= clk_25;



    -- Lógica de barrido horizontal y vertical (Tiempos estándar VGA)

    process(clk_25, reset)

    begin

        if reset = '1' then

            h_cnt <= (others => '0');

            v_cnt <= (others => '0');

        elsif rising_edge(clk_25) then

            if h_cnt = 799 then

                h_cnt <= (others => '0');

                if v_cnt = 524 then

                    v_cnt <= (others => '0');

                else

                    v_cnt <= v_cnt + 1;

                end if;

            else

                h_cnt <= h_cnt + 1;

            end if;

        end if;

    end process;



    -- Generación de señales de sincronización (Polaridad activa baja)

    hsync <= '0' when (h_cnt >= 656 and h_cnt <= 751) else '1';

    vsync <= '0' when (v_cnt >= 490 and v_cnt <= 491) else '1';



    -- Exportar contadores al módulo GU

    hcount <= h_cnt;

    vcount <= v_cnt;

    

    -- Se asigna a '0' ya que la dirección de memoria final se procesa en el GU

    -- Se mantiene este puerto para asegurar compatibilidad con la estructura del paquete.

    dir <= (others => '0');



end Behavioral;
